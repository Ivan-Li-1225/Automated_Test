git init  //建立git資料夾

rm -rf .git  //刪除git資料夾

touch test.txt  // 新增檔案 (test.txt)

git status  //顯示版控狀態

git add <filename>　//將單一檔案加入暫存區 (索引)

git add . // 將工作區所有檔案加入暫存區 (索引)

.gitignore // 內含不需要 Git 版控的檔案名稱

git diff // 顯示現在工作區的檔案與上一版 / 暫存區的檔案的差異

git diff commit1 commit2  //比較兩個 commit 之間的不同之處

git commit -m "messenge"  //將暫存區(索引)的檔案全部放入本地數據庫。

====="messenge 是此次 commit 的說明"=====


git log  // 顯示 commit 紀錄

git log --oneline  //顯示 commit 紀錄 (simple)

git checkout commitID //回溯版本 (全碼或前七碼都可以)

git checkout master  //切回到現在版本 

git pull --rebase origin main  // 整合github雲端數據庫與本地數據庫資料

git push -u origin main  // 將資料上傳至github雲端數據庫
